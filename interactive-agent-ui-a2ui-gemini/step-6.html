6. Teaching Gemini A2UI (Prompt-First Catalog)

In this step, you will update the backend agent to generate declarative A2UI components using the Prompt-First technique.

How Prompt-First A2UI Works

Instead of hardcoding frontend components or relying on custom SDKs, we teach Gemini the A2UI Home Automation Catalog directly in the system instructions. We instruct the model to follow a clear decision policy:

  • Conversational queries (e.g. "What humidity is ideal for sleep?"): The model responds in friendly Markdown text.
  • Device status and control requests (e.g. "Show me my dashboard" or "What's the status of the house?"): The model outputs a brief natural language greeting followed by an A2UI declarative JSON block defining the view tree.

def execute_smart_home_action(name: str, parameters: Dict[str, Any]) -> str:
    if name == "set_thermostat":
        temp = int(parameters.get("temperature", 72))
        mode = parameters.get("mode", "heat")
        SMART_HOME_STATE["thermostat"]["target_temperature"] = temp
        SMART_HOME_STATE["thermostat"]["mode"] = mode
        return f"Thermostat updated to {temp}°F ({mode} mode)."
    elif name == "set_light":
        room = parameters.get("room", "living_room")
        state = parameters.get("state", "on")
        if room in SMART_HOME_STATE["lights"]:
            SMART_HOME_STATE["lights"][room]["state"] = state
            SMART_HOME_STATE["lights"][room]["brightness"] = 100 if state == "on" else 0
            return f"{SMART_HOME_STATE['lights'][room]['name']} turned {state}."
    elif name == "set_door_lock":
        dev = parameters.get("device", "front_door")
        state = parameters.get("state", "locked")
        if dev in SMART_HOME_STATE["security"]:
            SMART_HOME_STATE["security"][dev]["state"] = state
            return f"{SMART_HOME_STATE['security'][dev]['name']} is now {state}."
    return "Action processed."

A2UI_SYSTEM_INSTRUCTION = """You are an intelligent Smart Home Voice & Chat Assistant equipped with A2UI (Agent-to-User Interface) generative UI capabilities.
You help users monitor and manage their IoT devices: thermostat, smart lights, indoor air quality, and security locks.

You have access to the `get_device_status` tool to check real-time device values.

IMPORTANT DECISION RULE FOR OUTPUT FORMAT:
You must decide between TWO response formats based on what the user asks:

1. TEXT-ONLY RESPONSE:
   When the user asks conversational questions, general knowledge, explanations, recommendations, or troubleshooting (e.g., "What humidity is ideal for sleep?", "How can I conserve energy with smart lights?", "How does a heat pump work?"):
   -> Respond ONLY in friendly, clear markdown text. Do NOT emit A2UI components or code blocks for informational questions.

2. A2UI GENERATIVE UI RESPONSE:
   When the user asks to see device status, inspect devices, show a dashboard, or adjust smart home controls (e.g., "Show me my smart home status", "Show the living room", "List my devices", "Thermostat settings", "Turn off living room light"):
   -> First provide a short friendly text summary (1-2 sentences).
   -> THEN append an A2UI declarative JSON block enclosed strictly in ```a2ui ... ``` containing the A2UI surface definition.

A2UI HOME AUTOMATION COMPONENT CATALOG:
- "Column": Vertically stacks child components. Children: array of component IDs.
- "Row": Horizontally stacks child components. Properties: {"alignment": "start"|"space-between"|"center"}. Children: array of IDs.
- "Card": Visual container with card styling and padding. Children: array of component IDs.
- "Text": Displays text. Properties: {"text": string, "variant": "title"|"subtitle"|"body"|"caption"}.
- "StatusBadge": Visual pill badge. Properties: {"text": string, "status": "normal"|"active"|"warning"|"offline"}.
- "Button": Clickable button that emits an action. Properties: {
    "label": string,
    "variant": "filled"|"outlined"|"tonal",
    "action": {"name": "set_thermostat"|"set_light"|"set_door_lock", "parameters": { ... }}
  }
Always ensure valid JSON inside the a2ui code block.
"""

def parse_a2ui_response(raw_text: str) -> Tuple[str, Optional[Dict[str, Any]]]:
    """Extracts text and optional A2UI declarative surface JSON from the model's response."""
    a2ui_match = re.search(r"```(?:a2ui|json)\s*\n([\s\S]*?)\n```", raw_text)
    if a2ui_match:
        json_content = a2ui_match.group(1).strip()
        text_before = raw_text[:a2ui_match.start()].strip()
        text_after = raw_text[a2ui_match.end():].strip()
        cleaned_text = f"{text_before} {text_after}".strip()

        try:
            surface_data = json.loads(json_content)
            if "components" in surface_data or "surfaceId" in surface_data:
                return cleaned_text, surface_data
        except json.JSONDecodeError:
            pass

    try:
        data = json.loads(raw_text.strip())
        if isinstance(data, dict):
            if "components" in data:
                return "Here is your smart home status:", data
            if "a2ui" in data and isinstance(data["a2ui"], dict):
                return data.get("text", ""), data["a2ui"]
    except json.JSONDecodeError:
        pass

    return raw_text.strip(), None

class ChatMessage(BaseModel):
    role: str
    content: str

class ChatRequest(BaseModel):
    message: str
    history: Optional[List[ChatMessage]] = None

class ChatResponse(BaseModel):
    text: str
    a2ui: Optional[Dict[str, Any]] = None

class ActionRequest(BaseModel):
    action: str
    parameters: Dict[str, Any]

class ActionResponse(BaseModel):
    message: str
    a2ui: Dict[str, Any]

@app.post("/api/chat", response_model=ChatResponse)
async def chat_endpoint(req: ChatRequest):
    try:
        contents = []
        if req.history:
            for msg in req.history:
                contents.append(types.Content(
                    role=msg.role,
                    parts=[types.Part.from_text(text=msg.content)]
                ))
        contents.append(types.Content(
            role="user",
            parts=[types.Part.from_text(text=req.message)]
        ))

        config = types.GenerateContentConfig(
            system_instruction=A2UI_SYSTEM_INSTRUCTION,
            temperature=0.3,
            tools=[get_device_status],
        )

        response = client.models.generate_content(
            model="gemini-3.8-flash",
            contents=contents,
            config=config,
        )

        raw_output = response.text or ""
        text_part, a2ui_surface = parse_a2ui_response(raw_output)

        # Fallback to current live dashboard if user asked for controls
        keywords = ["device", "status", "dashboard", "thermostat", "lights", "turn", "set", "home"]
        if not a2ui_surface and any(k in req.message.lower() for k in keywords) and "what is" not in req.message.lower() and "how" not in req.message.lower():
            a2ui_surface = build_a2ui_dashboard_surface()

        return ChatResponse(text=text_part or "Command executed.", a2ui=a2ui_surface)
    except Exception as e:
        print(f"Error calling Gemini: {e}")
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/api/action", response_model=ActionResponse)
async def action_endpoint(req: ActionRequest):
    msg = execute_smart_home_action(req.action, req.parameters)
    updated_surface = build_a2ui_dashboard_surface()
    return ActionResponse(message=msg, a2ui=updated_surface)
