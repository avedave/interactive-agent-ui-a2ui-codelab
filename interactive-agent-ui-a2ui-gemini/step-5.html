Understanding A2UI Protocol Components

In A2UI, an interface is defined as a JSON document containing:

  • surfaceId: An identifier for the UI container.
  • rootComponentId: The entry component of the view tree.
  • components: An array (adjacency list) of component definitions.

Each component specifies:

  • id: A unique string identifier.
  • component: The component type name from the catalog (e.g., Card, Row, Column, Text, StatusBadge, Button).
  • properties: Component-specific attributes (text, status colors, variants).
  • children: An array of child component IDs.
  • action: An event payload (for interactive buttons) detailing the action name and parameters.

Adding the Zero-Build Web Component Renderer (static/a2ui-bridge.js)

In traditional frontend workflows, integrating component libraries often requires npm, Webpack, or Vite. With A2UI, because components are rendered natively by Web Components, we can load and register <a2ui-surface> directly using browser ES modules.

Create static/a2ui-bridge.js:

cat << 'EOF' > static/a2ui-bridge.js
/**
 * A2UI Client Bridge & Web Component Renderer
 * Implements the Agent-to-User Interface (A2UI) declarative component protocol
 * for Web Components, enabling rich generative UI inside chat applications.
 */

class A2UISurface extends HTMLElement {
  constructor() {
    super();
    this.attachShadow({ mode: 'open' });
    this.componentsMap = new Map();
    this.rootId = null;
  }

  connectedCallback() {
    this.renderLoading();
  }

  renderLoading() {
    if (!this.shadowRoot.innerHTML) {
      this.shadowRoot.innerHTML = `
        <style>
          :host { display: block; margin: 0.5rem 0; }
          .loading { color: #5f6368; font-size: 0.85rem; padding: 0.5rem; }
        </style>
        <div class="loading">Loading interface...</div>
      `;
    }
  }

  /**
   * Main entry point to update the surface with an A2UI declarative payload.
   */
  updateSurface(surfaceData) {
    if (!surfaceData || !surfaceData.components) return;

    this.componentsMap.clear();
    for (const comp of surfaceData.components) {
      this.componentsMap.set(comp.id, comp);
    }

    this.rootId = surfaceData.rootComponentId || surfaceData.components[0]?.id;
    this.render();
  }

  render() {
    if (!this.rootId || !this.componentsMap.has(this.rootId)) return;

    const styles = `
      :host {
        display: block;
        font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, Arial, sans-serif;
        color: #202124;
      }
      .a2ui-column { display: flex; flex-direction: column; gap: 0.75rem; width: 100%; }
      .a2ui-row { display: flex; flex-direction: row; align-items: center; flex-wrap: wrap; gap: 0.5rem; }
      .a2ui-row.align-space-between { justify-content: space-between; }
      .a2ui-row.align-start { justify-content: flex-start; }

      .a2ui-card {
        background: #ffffff;
        border: 1px solid #dadce0;
        border-radius: 12px;
        padding: 1rem 1.25rem;
        box-shadow: 0 1px 3px rgba(60, 64, 67, 0.08);
        display: flex;
        flex-direction: column;
        gap: 0.75rem;
      }

      .a2ui-text { line-height: 1.4; }
      .a2ui-text.variant-title { font-size: 1.15rem; font-weight: 600; color: #1a73e8; }
      .a2ui-text.variant-subtitle { font-size: 1rem; font-weight: 600; color: #202124; }
      .a2ui-text.variant-body { font-size: 0.9rem; color: #3c4043; }
      .a2ui-text.variant-caption { font-size: 0.78rem; color: #70757a; }

      .a2ui-badge {
        font-size: 0.75rem;
        font-weight: 500;
        padding: 0.2rem 0.6rem;
        border-radius: 12px;
        display: inline-flex;
        align-items: center;
      }
      .a2ui-badge.status-active, .a2ui-badge.status-normal { background-color: #e6f4ea; color: #137333; }
      .a2ui-badge.status-warning { background-color: #fef7e0; color: #b06000; }
      .a2ui-badge.status-offline { background-color: #fce8e6; color: #c5221f; }

      .a2ui-button {
        padding: 0.4rem 0.9rem;
        border-radius: 8px;
        font-size: 0.85rem;
        font-weight: 500;
        cursor: pointer;
        border: none;
        display: inline-flex;
        align-items: center;
        gap: 0.4rem;
        transition: background-color 0.2s;
      }
      .a2ui-button.variant-filled { background-color: #1a73e8; color: #ffffff; }
      .a2ui-button.variant-filled:hover { background-color: #1557b0; }
      .a2ui-button.variant-outlined { background-color: transparent; border: 1px solid #dadce0; color: #1a73e8; }
      .a2ui-button.variant-tonal { background-color: #f1f3f4; color: #202124; }
    `;

    const rootElement = this.renderComponent(this.rootId);
    this.shadowRoot.innerHTML = `<style>${styles}</style>`;
    if (rootElement) {
      this.shadowRoot.appendChild(rootElement);
    }
  }

  renderComponent(componentId) {
    const comp = this.componentsMap.get(componentId);
    if (!comp) return null;

    const props = comp.properties || {};
    let el;

    switch (comp.component) {
      case 'Column': {
        el = document.createElement('div');
        el.className = 'a2ui-column';
        if (comp.children) {
          for (const childId of comp.children) {
            const childEl = this.renderComponent(childId);
            if (childEl) el.appendChild(childEl);
          }
        }
        break;
      }

      case 'Row': {
        el = document.createElement('div');
        const alignClass = props.alignment ? `align-${props.alignment}` : 'align-start';
        el.className = `a2ui-row ${alignClass}`;
        if (comp.children) {
          for (const childId of comp.children) {
            const childEl = this.renderComponent(childId);
            if (childEl) el.appendChild(childEl);
          }
        }
        break;
      }

      case 'Card': {
        el = document.createElement('div');
        el.className = 'a2ui-card';
        if (comp.children) {
          for (const childId of comp.children) {
            const childEl = this.renderComponent(childId);
            if (childEl) el.appendChild(childEl);
          }
        }
        break;
      }

      case 'Text': {
        el = document.createElement('div');
        const variant = props.variant || 'body';
        el.className = `a2ui-text variant-${variant}`;
        el.textContent = props.text || '';
        break;
      }

      case 'StatusBadge': {
        el = document.createElement('span');
        const status = props.status || 'normal';
        el.className = `a2ui-badge status-${status}`;
        el.textContent = props.text || '';
        break;
      }

      case 'Button': {
        el = document.createElement('button');
        el.className = `a2ui-button variant-${props.variant || 'filled'}`;
        el.textContent = props.label || 'Action';

        if (props.action) {
          el.addEventListener('click', (evt) => {
            evt.stopPropagation();
            this.dispatchEvent(new CustomEvent('a2ui-action', {
              detail: props.action,
              bubbles: true,
              composed: true,
            }));
          });
        }
        break;
      }

      default: {
        el = document.createElement('div');
        el.textContent = `[Unknown: ${comp.component}]`;
      }
    }

    return el;
  }
}

if (!customElements.get('a2ui-surface')) {
  customElements.define('a2ui-surface', A2UISurface);
}

export function setupA2UIBridge(container, onActionComplete) {
  container.addEventListener('a2ui-action', async (e) => {
    const action = e.detail;
    try {
      const response = await fetch('/api/action', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          action: action.name,
          parameters: action.parameters || {}
        })
      });

      if (!response.ok) throw new Error(`Action error: ${response.status}`);
      const result = await response.json();
      if (onActionComplete) onActionComplete(result);
    } catch (err) {
      alert(`Action error: ${err.message}`);
    }
  });
}

<!DOCTYPE html>
<html lang="en">
<head>

  <div class="toast-container" id="toastContainer"></div>

  <header>
    <div class="header-title">
      <h1>Smart Home Assistant</h1>
      <span class="badge-a2ui">✨ A2UI Generative UI</span>
    </div>
  </header>

  <main class="chat-container" id="chatContainer">
    <div class="message-wrapper assistant">
      <div class="message-sender">Gemini Assistant</div>
      <div class="message-bubble">
        Hello! I am your A2UI Smart Home Assistant. I can chat with you in natural language, or dynamically generate native interactive dashboards with live controls.
        <br><br>
        • Try asking: <em>"What is a healthy indoor humidity?"</em> (I'll reply in friendly text advice).<br>
        • Or try: <em>"Show me my home dashboard"</em> (I'll render an interactive A2UI surface with live controls).
      </div>
    </div>
  </main>

  <footer class="composer-container">
    <div class="quick-prompts">
      <button class="quick-chip" onclick="sendQuickPrompt('Show me my home dashboard')">🏠 Show home dashboard</button>
      <button class="quick-chip" onclick="sendQuickPrompt('Turn off the living room light')">💡 Turn off living room</button>
      <button class="quick-chip" onclick="sendQuickPrompt('What is a healthy indoor humidity?')">❓ Healthy humidity?</button>
      <button class="quick-chip" onclick="sendQuickPrompt('How can I save electricity with heating?')">⚡ Energy saving tips</button>
    </div>
    <form class="composer-form" id="chatForm">
      <input type="text" id="userInput" class="composer-input" placeholder="Ask a question or request controls..." autocomplete="off" required>
      <button type="submit" id="sendBtn" class="composer-button">Send</button>
    </form>
  </footer>

  <script type="module">
    import { setupA2UIBridge } from '/static/a2ui-bridge.js';

    const chatContainer = document.getElementById('chatContainer');
    const chatForm = document.getElementById('chatForm');
    const userInput = document.getElementById('userInput');
    const sendBtn = document.getElementById('sendBtn');
    const toastContainer = document.getElementById('toastContainer');

    const history = [];
    let currentActiveSurface = null;

    function showToast(message) {
      const toast = document.createElement('div');
      toast.className = 'toast';
      toast.textContent = message;
      toastContainer.appendChild(toast);
      setTimeout(() => toast.remove(), 3000);
    }

    // Connect the A2UI bridge to handle action button clicks
    setupA2UIBridge(document.body, (actionResult) => {
      showToast(`⚡ ${actionResult.message}`);
      if (currentActiveSurface && actionResult.a2ui) {
        currentActiveSurface.updateSurface(actionResult.a2ui);
      }
    });

    function appendMessage(role, text, a2uiSurfaceData = null) {
      const wrapper = document.createElement('div');
      wrapper.className = `message-wrapper ${role}`;

      const sender = document.createElement('div');
      sender.className = 'message-sender';
      sender.textContent = role === 'user' ? 'You' : 'Gemini Assistant (A2UI)';

      const bubble = document.createElement('div');
      bubble.className = 'message-bubble';
      if (text) bubble.textContent = text;

      wrapper.appendChild(sender);
      wrapper.appendChild(bubble);

      // Mount <a2ui-surface> when an A2UI payload is returned
      if (a2uiSurfaceData) {
        const surfaceContainer = document.createElement('div');
        surfaceContainer.className = 'surface-wrapper';

        const surfaceEl = document.createElement('a2ui-surface');
        surfaceEl.setAttribute('surface-id', a2uiSurfaceData.surfaceId || 'home-dashboard');

        surfaceContainer.appendChild(surfaceEl);
        bubble.appendChild(surfaceContainer);

        surfaceEl.updateSurface(a2uiSurfaceData);
        currentActiveSurface = surfaceEl;
      }

      chatContainer.appendChild(wrapper);
      chatContainer.scrollTop = chatContainer.scrollHeight;
      return bubble;
    }

    async function handleSendMessage(messageText) {
      if (!messageText.trim()) return;

      appendMessage('user', messageText);
      history.push({ role: 'user', content: messageText });
      userInput.value = '';
      sendBtn.disabled = true;

      const loadingBubble = appendMessage('assistant', 'Thinking...');

      try {
        const response = await fetch('/api/chat', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ message: messageText, history: history.slice(0, -1) })
        });

        if (!response.ok) throw new Error(`Server error ${response.status}`);
        const data = await response.json();

        loadingBubble.closest('.message-wrapper').remove();
        appendMessage('assistant', data.text, data.a2ui);
        history.push({ role: 'model', content: data.text });
      } catch (err) {
        loadingBubble.textContent = `⚠️ Error: ${err.message}`;
      } finally {
        sendBtn.disabled = false;
        userInput.focus();
      }
    }

    chatForm.addEventListener('submit', (e) => {
      e.preventDefault();
      handleSendMessage(userInput.value);
    });

    window.sendQuickPrompt = function(text) {
      userInput.value = text;
      handleSendMessage(text);
    };
  </script>
</body>
</html>
