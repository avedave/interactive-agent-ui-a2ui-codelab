3. Building the Baseline Smart Home Agent

Before introducing A2UI, you will build a functional baseline: a Python FastAPI server running gemini-3.8-flash that interacts with a simulated smart home in memory and serves a simple web chat interface.

Create the Backend (main.py)

In ~/a2ui-codelab/01-baseline, create main.py:

cat << 'EOF' > main.py
import os
from typing import List, Optional
from fastapi import FastAPI, HTTPException
from fastapi.staticfiles import StaticFiles
from fastapi.responses import FileResponse
from pydantic import BaseModel
from google import genai
from google.genai import types

app = FastAPI(title="Smart Home Assistant (Baseline)")

# In-memory device state for home automation
SMART_HOME_STATE = {
    "thermostat": {
        "location": "Living Room",
        "current_temperature": 71,
        "target_temperature": 72,
        "humidity": 45,
        "mode": "heat",
        "battery": 92,
    },
    "lights": {
        "living_room": {"name": "Living Room Ceiling", "state": "on", "brightness": 80},
        "kitchen": {"name": "Kitchen Pendant", "state": "off", "brightness": 0},
        "bedroom": {"name": "Bedroom Lamp", "state": "on", "brightness": 50},
    },
    "air_quality": {
        "location": "Nursery",
        "co2": 540,
        "status": "Good",
        "humidity": 48,
        "battery": 85,
    },
    "security": {
        "front_door": {"name": "Front Door Lock", "state": "locked", "battery": 94},
        "garage": {"name": "Garage Door", "state": "closed", "battery": 90},
    },
}

# Python functions exposed as tools for Gemini
def get_device_status() -> dict:
    """Retrieves the current real-time status of all smart home devices including thermostat, lights, air quality, and security."""
    return SMART_HOME_STATE

def set_thermostat(temperature: int, mode: str = "heat") -> str:
    """Sets the thermostat target temperature (in Fahrenheit) and operating mode (heat, cool, eco, off)."""
    SMART_HOME_STATE["thermostat"]["target_temperature"] = temperature
    SMART_HOME_STATE["thermostat"]["mode"] = mode
    return f"Thermostat target set to {temperature}°F in {mode} mode."

def set_light(room: str, state: str, brightness: int = 100) -> str:
    """Controls a light. Room must be 'living_room', 'kitchen', or 'bedroom'. State must be 'on' or 'off'. Brightness is 0-100."""
    if room not in SMART_HOME_STATE["lights"]:
        return f"Unknown room '{room}'. Valid rooms: {list(SMART_HOME_STATE['lights'].keys())}"
    SMART_HOME_STATE["lights"][room]["state"] = state
    SMART_HOME_STATE["lights"][room]["brightness"] = brightness if state == "on" else 0
    return f"{SMART_HOME_STATE['lights'][room]['name']} turned {state} (brightness {brightness}%)."

def set_door_lock(device: str, state: str) -> str:
    """Controls locks or doors. Device must be 'front_door' or 'garage'. State must be 'locked', 'unlocked', 'closed', or 'open'."""
    if device not in SMART_HOME_STATE["security"]:
        return f"Unknown security device '{device}'."
    SMART_HOME_STATE["security"][device]["state"] = state
    return f"{SMART_HOME_STATE['security'][device]['name']} is now {state}."

# Configure Vertex AI Gemini client
PROJECT_ID = os.environ.get("GOOGLE_CLOUD_PROJECT") or os.environ.get("GCP_PROJECT")
LOCATION = os.environ.get("GOOGLE_CLOUD_REGION", "us-central1")

client = genai.Client(
    vertexai=True,
    project=PROJECT_ID,
    location=LOCATION,
)

SYSTEM_INSTRUCTION = """You are a helpful Smart Home Voice & Chat Assistant.
You help users monitor and control their smart home devices: thermostat, lights, air quality sensors, and door locks.
You have access to tools to read the device status and modify device settings.
Always provide clear, friendly responses. When summarizing device status or changes, present the details clearly in text.
If the user asks questions unrelated to home devices (such as energy tips or comfort advice), answer naturally and helpfully.
"""

class ChatMessage(BaseModel):
    role: str
    content: str

class ChatRequest(BaseModel):
    message: str
    history: Optional[List[ChatMessage]] = None

class ChatResponse(BaseModel):
    response: str

@app.post("/api/chat", response_model=ChatResponse)
async def chat_endpoint(req: ChatRequest):
    try:
        contents = []
        if req.history:
            for msg in req.history:
                contents.append(types.Content(
                    role=msg.role,
                    parts=[types.Part.from_text(text=msg.content)]
                ))
        contents.append(types.Content(
            role="user",
            parts=[types.Part.from_text(text=req.message)]
        ))

        config = types.GenerateContentConfig(
            system_instruction=SYSTEM_INSTRUCTION,
            temperature=0.4,
            tools=[get_device_status, set_thermostat, set_light, set_door_lock],
        )

        response = client.models.generate_content(
            model="gemini-3.8-flash",
            contents=contents,
            config=config,
        )

        return ChatResponse(response=response.text or "Command executed.")
    except Exception as e:
        print(f"Error calling Gemini: {e}")
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/api/devices")
async def get_devices():
    return SMART_HOME_STATE

# Mount static frontend
static_dir = os.path.join(os.path.dirname(__file__), "static")
if os.path.exists(static_dir):
    app.mount("/static", StaticFiles(directory=static_dir), name="static")

@app.get("/")
async def root():
    index_file = os.path.join(static_dir, "index.html")
    if os.path.exists(index_file):
        return FileResponse(index_file)
    return {"message": "Smart Home Assistant API is running."}

if __name__ == "__main__":
    import uvicorn
    port = int(os.environ.get("PORT", 8080))
    uvicorn.run("main:app", host="0.0.0.0", port=port, reload=True)
EOF

Create static/index.html:

cat << 'EOF' > static/index.html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Smart Home Assistant (Baseline)</title>
  <link rel="stylesheet" href="/static/styles.css">
</head>
<body>
  <header>
    <div class="header-title">
      <h1>Smart Home Assistant</h1>
      <span class="badge">Baseline: Text-Only</span>
    </div>
  </header>

  <main class="chat-container" id="chatContainer">
    <div class="message-wrapper assistant">
      <div class="message-sender">Gemini Assistant</div>
      <div class="message-bubble">Hello! I am your Smart Home Assistant. I can monitor your smart devices and adjust settings. Try asking "Show all device statuses" or "Turn off the kitchen lights".</div>
    </div>
  </main>

  <footer class="composer-container">
    <div class="quick-prompts">
      <button class="quick-chip" onclick="sendQuickPrompt('Show all device statuses')">📊 Show all device statuses</button>
      <button class="quick-chip" onclick="sendQuickPrompt('Turn off the living room light')">💡 Turn off living room light</button>
      <button class="quick-chip" onclick="sendQuickPrompt('Set thermostat to 70')">🌡️ Set thermostat to 70°F</button>
      <button class="quick-chip" onclick="sendQuickPrompt('What is a healthy indoor humidity?')">❓ Healthy indoor humidity?</button>
    </div>
    <form class="composer-form" id="chatForm">
      <input type="text" id="userInput" class="composer-input" placeholder="Ask about devices or give a command..." autocomplete="off" required>
      <button type="submit" id="sendBtn" class="composer-button">Send</button>
    </form>
  </footer>

  <script>
    const chatContainer = document.getElementById('chatContainer');
    const chatForm = document.getElementById('chatForm');
    const userInput = document.getElementById('userInput');
    const sendBtn = document.getElementById('sendBtn');
    const history = [];

    function appendMessage(role, text) {
      const wrapper = document.createElement('div');
      wrapper.className = `message-wrapper ${role}`;

      const sender = document.createElement('div');
      sender.className = 'message-sender';
      sender.textContent = role === 'user' ? 'You' : 'Gemini Assistant';

      const bubble = document.createElement('div');
      bubble.className = 'message-bubble';
      bubble.textContent = text;

      wrapper.appendChild(sender);
      wrapper.appendChild(bubble);
      chatContainer.appendChild(wrapper);
      chatContainer.scrollTop = chatContainer.scrollHeight;
      return bubble;
    }

    async function handleSendMessage(messageText) {
      if (!messageText.trim()) return;

      appendMessage('user', messageText);
      history.push({ role: 'user', content: messageText });
      userInput.value = '';
      sendBtn.disabled = true;

      const loadingBubble = appendMessage('assistant', 'Thinking...');

      try {
        const response = await fetch('/api/chat', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ message: messageText, history: history.slice(0, -1) })
        });

        if (!response.ok) {
          const err = await response.json().catch(() => ({ detail: 'Network error' }));
          throw new Error(err.detail || `Server error ${response.status}`);
        }

        const data = await response.json();
        loadingBubble.textContent = data.response;
        history.push({ role: 'model', content: data.response });
      } catch (err) {
        loadingBubble.textContent = `⚠️ Error: ${err.message}`;
      } finally {
        sendBtn.disabled = false;
        userInput.focus();
      }
    }

    chatForm.addEventListener('submit', (e) => {
      e.preventDefault();
      handleSendMessage(userInput.value);
    });

    function sendQuickPrompt(text) {
      userInput.value = text;
      handleSendMessage(text);
    }
  </script>
</body>
</html>
EOF

Test the Baseline Locally

Start the application using uvicorn:

uvicorn main:app --host 0.0.0.0 --port 8080

In Google Cloud Shell, click the Web Preview icon (eye icon in the top right) and select Preview on port 8080.

Try the following interactions:

  • Click "❓ Healthy indoor humidity?"
    • The model responds with informative conversational text.
  • Click "📊 Show all device statuses"
    • The model calls get_device_status and outputs a long, bulleted text summary of all devices.

Friction Alert: When inspecting multiple devices, reading paragraphs of text is tedious. If you want to change the temperature or turn off a light, you have to type another command. There are no toggles, no sliders, and no visual indicators.

Stop the server in the terminal with Ctrl+C.
